{{ page.title }}

Prepare

no swap

swapoff -a

system module

cat <<EOF | sudo tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
nfnetlink_acct
EOF

system config

cat <<EOF | sudo tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

(optional)

export NETAVARK_FW=iptables


Install dependency

apt update
apt install -y curl apt-transport-https ca-certificates gnupg


Install

keyring and source

export KUBERNETES_VERSION=v1.32
curl -fsSL https://pkgs.k8s.io/core:/stable:/$KUBERNETES_VERSION/deb/Release.key | sudo gpg —dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
echo "deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/$KUBERNETES_VERSION/deb/ /" | sudo tee /etc/apt/sources.list.d/kubernetes.list

install

apt update
apt install  kubectl kubeadm kubelet cri-tools
systemctl enable kubelet --now

(optional) for master

/*** calico ***/
kubectl apply -f calico.yaml
/*** set IP explicitly ***/
/etc/default/kubelet
	KUBELET_EXTRA_ARGS="--node-ip=10.0.0.21"

config

/var/lib/kubelet/config.yaml
	cgroupDriver: system
	serverTLSBootstrap: true

(optional) for crictl to change CRI

/etc/crictl.yaml
	runtime-endpoint: "unix:///var/run/containerd/containerd.sock"

certificate

kubectl get csr
	kubectl certificate approve [csr-name]

check

systemctl status kubelet
kubelet --version


Reset

kubeadm reset --force
rm -rf /etc/kubernetes /var/lib/kubelet /var/lib/etcd
rm -rf /etc/cni/net.d /var/lib/cni /var/lib/calico/
//(optional) rm -rf $HOME/.kube



Worker

get config

scp username@10.0.0.21:.kube/config ~/.kube/config

join

$(kubeadm token create --print-join-command) --cri-socket=unix:///var/run/crio/crio.sock --node-name [node name]

check iptable configuration

curl -k https://10.96.0.1:443/healthz

FIX:

sudo update-alternatives --set iptables /usr/sbin/iptables-legacy
sudo update-alternatives --set ip6tables /usr/sbin/ip6tables-legacy

kubectl set env daemonset/calico-node -n kube-system IP_AUTODETECTION_METHOD=can-reach=10.0.0.21
kubectl set env daemonset/calico-node -n kube-system FELIX_IPTABLESBACKEND=Legacy
