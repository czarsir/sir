kube-proxy

  • ConfigMap
    • kubectl edit configmap -n kube-system kube-proxy
data:
  config.conf:
    iptables:
      # Force legacy iptables mode
      localhostNodePorts: true
  • DaemonSet
    • kubectl edit daemonset -n kube-system kube-proxy
	//command
	//	--proxy-mode=nftables


calico-node

  • DaemonSet
    • kubectl edit ds -n kube-system calico-node
	- name: IP_AUTODETECTION_METHOD
	  value: can-reach=10.0.0.21
kubectl set env daemonset/calico-node -n kube-system IP_AUTODETECTION_METHOD=can-reach=10.0.0.21
kubectl set env daemonset/calico-node -n kube-system FELIX_IPTABLESBACKEND=Legacy


coredns

  • ConfigMap
    • kubectl edit configmap -n kube-system coredns
	from:	forward . /etc/resolv.conf
	to:	forward . [dns server IP]
kubectl rollout restart deployment coredns -n kube-system
