Install

keyring and source

curl -fsSL https://download.docker.com/linux/debian/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg --yes
echo "deb [arch=arm64 signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/debian trixie stable" | sudo tee /etc/apt/sources.list.d/docker.list

install

sudo apt update
sudo apt install -y containerd.io

(optional) downgrade

rm -rf /etc/containerd  /var/lib/containerd
apt install -y --allow-downgrades containerd.io=1.7.28-1~debian.13~trixie
systemctl restart containerd
systemctl restart kubelet

create default config

containerd config default | sudo tee /etc/containerd/config.toml

edit config

sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml
systemctl restart containerd

check

systemctl status containerd


Config

/etc/containerd/config.toml (mirror)

[plugins."io.containerd.grpc.v1.cri".registry.mirrors]
	[plugins."io.containerd.grpc.v1.cri".registry.mirrors."docker.io"]
		endpoint = ["https://your-mirror-address.com", "https://registry-1.docker.io"]


Tools

ctr

ctr -n k8s.io images pull registry.aliyuncs.com/google_containers/pause:3.8
ctr -n k8s.io image import my_image.tar
ctr --address /var/run/containerd/containerd.sock -n k8s.io image ls
